import { clampYear } from "./yearRange";

export interface ZoomState {
  readonly level: "overview" | "month";
  readonly year: number;
  readonly month?: number;
}

export function parseZoom(params: URLSearchParams, fallbackYear: number): ZoomState {
  const yRaw = params.get("y");
  const yNum = yRaw !== null ? Number(yRaw) : NaN;
  const year = clampYear(Number.isFinite(yNum) ? Math.trunc(yNum) : fallbackYear);

  const mRaw = params.get("m");
  const mNum = mRaw !== null ? Number(mRaw) : NaN;
  if (Number.isInteger(mNum) && mNum >= 1 && mNum <= 12) {
    return { level: "month", year, month: mNum };
  }
  return { level: "overview", year };
}

export function zoomToQuery(state: ZoomState): string {
  const base = `?y=${state.year}`;
  return state.level === "month" && state.month !== undefined
    ? `${base}&m=${state.month}`
    : base;
}
```

"use client";

import { useCallback, useEffect, useRef, useState } from "react";
import { flushSync } from "react-dom";
import { parseZoom, zoomToQuery, type ZoomState } from "@/domain/view";

interface CivilDate {
  readonly year: number;
  readonly month: number;
  readonly day: number;
}

type HistoryMode = "push" | "replace" | "none";

function withTransition(commit: () => void): void {
  const reduce =
    typeof window !== "undefined" &&
    window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  const canVT =
    typeof document !== "undefined" &&
    "startViewTransition" in document;
  if (canVT && !reduce) {
    (document as unknown as { startViewTransition: (cb: () => void) => void })
      .startViewTransition(() => flushSync(commit));
  } else {
    commit();
  }
}

export function useZoomState() {
  const [today, setToday] = useState<CivilDate | null>(null);
  const [zoom, setZoom] = useState<ZoomState>(() => ({
    level: "overview",
    year: new Date().getFullYear(),
  }));
  const fallbackYearRef = useRef<number>(new Date().getFullYear());

  useEffect(() => {
    const now = new Date();
    const t: CivilDate = { year: now.getFullYear(), month: now.getMonth() + 1, day: now.getDate() };
    setToday(t);
    fallbackYearRef.current = t.year;
    setZoom(parseZoom(new URLSearchParams(window.location.search), t.year));
  }, []);

  useEffect(() => {
    const onPop = (): void => {
      const next = parseZoom(new URLSearchParams(window.location.search), fallbackYearRef.current);
      withTransition(() => setZoom(next));
    };
    window.addEventListener("popstate", onPop);
    return () => window.removeEventListener("popstate", onPop);
  }, []);

  const navigate = useCallback((next: ZoomState, mode: HistoryMode): void => {
    withTransition(() => {
      setZoom(next);
      const url = zoomToQuery(next);
      if (mode === "push") window.history.pushState({}, "", url);
      else if (mode === "replace") window.history.replaceState({}, "", url);
    });
  }, []);

  return { zoom, today, navigate };
}
```
