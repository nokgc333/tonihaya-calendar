**Goal:** L1(月)の日をクリックして、その日を含む週を Google Calendar 風の時間グリッド（7日×0-23時・今日/祝日/土休/現在時刻ライン）で表示する L2 週間ビューを追加する。

**Tech Stack:** Next.js 15(App Router, static export) / React 19 / TypeScript strict / Tailwind v4 / Vitest / Playwright。既存 domain(calendar/holiday/view/shared) を再利用。

```ts
import { clampYear } from "./yearRange";

export interface ZoomState {
  readonly level: "overview" | "month" | "week" | "day";
  readonly year: number;
  readonly month?: number;
  readonly day?: number;
}

function intInRange(raw: string | null, min: number, max: number): number | null {
  if (raw === null) return null;
  const n = Number(raw);
  if (!Number.isInteger(n) || n < min || n > max) return null;
  return n;
}

export function parseZoom(params: URLSearchParams, fallbackYear: number): ZoomState {
  const yRaw = params.get("y");
  const yNum = yRaw !== null ? Number(yRaw) : NaN;
  const year = clampYear(Number.isFinite(yNum) ? Math.trunc(yNum) : fallbackYear);

  const month = intInRange(params.get("m"), 1, 12);
  const day = intInRange(params.get("d"), 1, 31);
  const v = params.get("v");

  if (month !== null && day !== null && (v === "week" || v === "day")) {
    return { level: v, year, month, day };
  }
  if (month !== null) {
    return { level: "month", year, month };
  }
  return { level: "overview", year };
}

export function zoomToQuery(state: ZoomState): string {
  const base = `?y=${state.year}`;
  if (state.level === "overview") return base;
  if (state.level === "month") return `${base}&m=${state.month}`;
  return `${base}&m=${state.month}&d=${state.day}&v=${state.level}`;
}
```

```ts
import { getHoliday } from "../holiday";
import { fromEpochDay, toEpochDay, weekdayOf } from "../shared/date";
import type { WeekStart } from "../calendar";

const DAYS_PER_WEEK = 7;
const SUNDAY = 0;
const SATURDAY = 6;
const MINUTES_PER_HOUR = 60;

interface CivilDate {
  readonly year: number;
  readonly month: number;
  readonly day: number;
}

export interface WeekDay {
  readonly year: number;
  readonly month: number;
  readonly day: number;
  readonly weekday: number;
  readonly isToday: boolean;
  readonly isHoliday: boolean;
  readonly holidayName?: string;
  readonly isSaturday: boolean;
  readonly isSunday: boolean;
}

export interface WeekViewModel {
  readonly days: readonly WeekDay[];
}

/** 基準日を含む週（週開始曜日から7日）を組み立てる。 */
export function buildWeekView(
  year: number,
  month: number,
  day: number,
  today: CivilDate | null,
  settings: { weekStart: WeekStart },
): WeekViewModel {
  const startOffset = settings.weekStart === "sun" ? 0 : 1;
  const refWeekday = weekdayOf(year, month, day);
  const back = (refWeekday - startOffset + DAYS_PER_WEEK) % DAYS_PER_WEEK;
  const startEpoch = toEpochDay(year, month, day) - back;

  const days: WeekDay[] = [];
  for (let i = 0; i < DAYS_PER_WEEK; i++) {
    const d = fromEpochDay(startEpoch + i);
    const weekday = weekdayOf(d.year, d.month, d.day);
    const holiday = getHoliday(d.year, d.month, d.day);
    days.push({
      year: d.year,
      month: d.month,
      day: d.day,
      weekday,
      isToday:
        today !== null &&
        today.year === d.year &&
        today.month === d.month &&
        today.day === d.day,
      isHoliday: holiday !== null,
      holidayName: holiday?.name,
      isSaturday: weekday === SATURDAY,
      isSunday: weekday === SUNDAY,
    });
  }
  return { days };
}

/** 0時からの経過分を、1時間=hourHeightPx としたときの縦オフセット(px)へ変換。 */
export function minutesToOffset(minutes: number, hourHeightPx: number): number {
  return (minutes / MINUTES_PER_HOUR) * hourHeightPx;
}
```

```tsx
"use client";

import { Fragment, useEffect, useRef } from "react";
import { buildWeekView, minutesToOffset, type WeekDay } from "@/domain/view";
import { fromEpochDay, toEpochDay } from "@/domain/shared/date";

interface CivilDate {
  readonly year: number;
  readonly month: number;
  readonly day: number;
}

const HOUR_HEIGHT = 48; // 1時間あたりの高さ(px)
const HOURS = Array.from({ length: 24 }, (_, h) => h);
const WEEKDAY_JA = ["日", "月", "火", "水", "木", "金", "土"] as const;
const GRID_COLS = "3.5rem repeat(7, minmax(4.5rem, 1fr))";
const SCROLL_LEAD_HOURS = 2;

function dayColor(d: WeekDay): string {
  if (d.isSunday || d.isHoliday) return "var(--accent-red)";
  if (d.isSaturday) return "var(--accent-sat)";
  return "var(--text)";
}
function dayBg(d: WeekDay): string {
  if (d.isSunday || d.isHoliday) return "var(--sun-bg)";
  if (d.isSaturday) return "var(--sat-bg)";
  return "var(--surface)";
}

export function WeekView({
  year,
  month,
  day,
  today,
  onBack,
  onWeekChange,
}: {
  readonly year: number;
  readonly month: number;
  readonly day: number;
  readonly today: CivilDate | null;
  readonly onBack: () => void;
  readonly onWeekChange: (year: number, month: number, day: number) => void;
}) {
  const week = buildWeekView(year, month, day, today, { weekStart: "sun" });
  const bodyRef = useRef<HTMLDivElement>(null);
  const todayIndex = week.days.findIndex((d) => d.isToday);

  useEffect(() => {
    const onKey = (e: KeyboardEvent): void => {
      if (e.key === "Escape") onBack();
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
  }, [onBack]);

  // 今日が週内なら現在時刻付近へ自動スクロール
  useEffect(() => {
    if (!bodyRef.current || todayIndex < 0) return;
    const now = new Date();
    const minutes = now.getHours() * 60 + now.getMinutes();
    bodyRef.current.scrollTop = Math.max(
      0,
      minutesToOffset(minutes, HOUR_HEIGHT) - HOUR_HEIGHT * SCROLL_LEAD_HOURS,
    );
  }, [todayIndex, year, month, day]);

  const shiftWeek = (deltaWeeks: number): void => {
    const next = fromEpochDay(toEpochDay(year, month, day) + deltaWeeks * 7);
    onWeekChange(next.year, next.month, next.day);
  };

  const first = week.days[0];
  const last = week.days[6];
  const nowOffset =
    todayIndex >= 0
      ? minutesToOffset(new Date().getHours() * 60 + new Date().getMinutes(), HOUR_HEIGHT)
      : 0;

  return (
    <div className="flex h-full flex-col">
      <nav className="mb-3 flex shrink-0 items-center gap-2">
        <button
          type="button"
          onClick={onBack}
          className="rounded-md border px-3 py-1 text-sm font-medium transition-opacity hover:opacity-70"
          style={{ borderColor: "var(--border)", color: "var(--text)" }}
          aria-label="月に戻る（Escでも戻れます）"
        >
          ‹ 戻る
        </button>
        <div className="ml-2 flex items-center gap-1.5">
          <button
            type="button"
            onClick={() => shiftWeek(-1)}
            className="rounded-md border px-2 py-1 text-sm transition-opacity hover:opacity-70"
            style={{ borderColor: "var(--border)", color: "var(--text)" }}
            aria-label="前の週"
          >
            ‹
          </button>
          <span className="min-w-[13rem] text-center text-base font-bold tabular-nums">
            {first.year}年{first.month}月{first.day}日 〜 {last.month}月{last.day}日
          </span>
          <button
            type="button"
            onClick={() => shiftWeek(1)}
            className="rounded-md border px-2 py-1 text-sm transition-opacity hover:opacity-70"
            style={{ borderColor: "var(--border)", color: "var(--text)" }}
            aria-label="次の週"
          >
            ›
          </button>
        </div>
      </nav>

      {/* スクロール横断の外枠。列幅を保つため横スクロール可 */}
      <div
        className="flex min-h-0 flex-1 flex-col overflow-x-auto rounded-md border"
        style={{ borderColor: "var(--border)" }}
      >
        {/* 曜日+日付ヘッダー（sticky） */}
        <div
          className="sticky top-0 z-10 grid shrink-0 border-b"
          style={{ gridTemplateColumns: GRID_COLS, borderColor: "var(--border)", background: "var(--surface)" }}
        >
          <div />
          {week.days.map((d) => (
            <div
              key={`h-${d.month}-${d.day}`}
              className="border-l p-1 text-center"
              style={{ borderColor: "var(--border)", background: dayBg(d) }}
            >
              <div className="text-[11px]" style={{ color: dayColor(d) }}>
                {WEEKDAY_JA[d.weekday]}
              </div>
              {d.isToday ? (
                <span
                  className="mx-auto flex h-7 w-7 items-center justify-center rounded-full text-base font-bold"
                  style={{ background: "var(--today-bg)", color: "var(--today-fg)" }}
                  aria-current="date"
                >
                  {d.day}
                </span>
              ) : (
                <div className="text-base font-bold tabular-nums" style={{ color: dayColor(d) }}>
                  {d.day}
                </div>
              )}
            </div>
          ))}
        </div>

        {/* 終日行（祝日名） */}
        <div
          className="grid shrink-0 border-b"
          style={{ gridTemplateColumns: GRID_COLS, borderColor: "var(--border)" }}
        >
          <div className="p-1 text-right text-[10px]" style={{ color: "var(--text-muted)" }}>
            終日
          </div>
          {week.days.map((d) => (
            <div
              key={`a-${d.month}-${d.day}`}
              className="min-h-[1.5rem] border-l p-0.5 text-center text-[10px]"
              style={{ borderColor: "var(--border)", background: dayBg(d), color: "var(--accent-red)" }}
            >
              {d.holidayName ?? ""}
            </div>
          ))}
        </div>

        {/* 時間グリッド本体（縦スクロール） */}
        <div ref={bodyRef} className="min-h-0 flex-1 overflow-y-auto">
          <div className="relative grid" style={{ gridTemplateColumns: GRID_COLS }}>
            {HOURS.map((h) => (
              <Fragment key={h}>
                <div
                  className="border-t pr-1 text-right text-[10px]"
                  style={{ height: HOUR_HEIGHT, borderColor: "var(--border)", color: "var(--text-muted)" }}
                >
                  {h}:00
                </div>
                {week.days.map((d) => (
                  <div
                    key={`c-${h}-${d.month}-${d.day}`}
                    className="border-t border-l"
                    style={{ height: HOUR_HEIGHT, borderColor: "var(--border)", background: dayBg(d) }}
                  />
                ))}
              </Fragment>
            ))}
            {/* 今ライン（今日の列のみ） */}
            {todayIndex >= 0 && (
              <div
                className="pointer-events-none absolute"
                style={{
                  top: nowOffset,
                  left: `calc(3.5rem + ${todayIndex} * (100% - 3.5rem) / 7)`,
                  width: `calc((100% - 3.5rem) / 7)`,
                  borderTop: "2px solid var(--accent-red)",
                }}
              />
            )}
          </div>
        </div>
      </div>
    </div>
  );
}
```
